import { Calendar, Droplets } from "lucide-react";
import * as LucideIcons from "lucide-react";
import { formatTemperature } from "../utils/index.js";

const WeatherForecast = ({ forecast, unit }) => {
  const dailyForecast = forecast.list.reduce((acc, curr) => {
    const date = new Date(curr.dt * 1000).toLocaleDateString("en-UK", {
      day: "numeric",
      weekday: "long",
      month: "long",
    });
    if (!acc[date]) {
      acc[date] = [];
    }
    acc[date].push(curr);
    return acc;
  }, {});

  const dailyItems = Object.values(dailyForecast).slice(1, 6);
  return (
    <div className="bg-white/10 backdrop-blur-xl border border-white/20 rounded-3xl p-8 shadow-2xl">
      <div className="flex items-center space-x-3 mb-8">
        <div className="p-2 bg-white/20 rounded-full">
          <Calendar className="w-6 h-6 text-white" />
        </div>
        <h2 className="text-2xl font-bold text-white">5 Day Forecast</h2>
      </div>
      <div className="space-y-4">
        {/* Map Method  */}
        {dailyItems.map((day, index) => {
          const date = new Date(day[0].dt * 1000).toLocaleDateString("en-UK", {
            day: "numeric",
            weekday: "long",
            month: "long",
          });
          const weatherDescription = day[0].weather[0].description;
          const iconName = day[0].weather[0].icon;
          const IconComponent = LucideIcons[iconName] || LucideIcons.Sun;

          const mainTemp = day[0].main.temp;
          const humidity = day[0].main.humidity;

          return (
            <div
              key={index}
              className="flex items-center justify-between p-5 bg-white/5 backdrop-blur-sm rounded-2xl hover:bg-white/10 transition-all duration-300 group border border-white/10"
            >
              <div className="flex items-center space-x-5 flex-1">
                <div className="text-white/90 group-hover:text-white transition-all transform group-hover:scale-110 duration">
                  <IconComponent className="w-8 h-8" />
                </div>
                <div className="flex-1">
                  <div className="text-white font-semibold text-lg">{date}</div>
                  <div className="text-white/70 text-sm capitalize font-medium">
                    {weatherDescription}
                  </div>
                </div>
              </div>
              <div className="flex items-center space-x-6">
                <div className="flex items-center space-x-2 text-white/60">
                  <Droplets className="w-4 h-4 text-blue-300" />
                  <span className="text-sm font-medium">{humidity}%</span>
                </div>
                <div className="text-right">
                  <div className="text-white font-bold text-lg">Main Temp</div>
                  <div className="text-white text-sm font-medium">
                    {formatTemperature(mainTemp, unit)}
                  </div>
                </div>
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
};

export default WeatherForecast;
